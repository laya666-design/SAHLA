rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAdmin() {
      return request.auth != null && request.auth.token.admin == true;
    }

    function isOwner(uid) {
      return request.auth != null && request.auth.uid == uid;
    }

    // Demandes de pièces
    match /part_requests/{requestId} {
      allow read: if true;
      allow create: if request.resource.data.statut == 'open';
      allow update: if isAdmin()
                    || (request.resource.data.diff(resource.data)
                          .affectedKeys()
                          .hasOnly(['statut', 'soldToStoreId', 'soldToStoreNom', 'dateVente'])
                        && request.resource.data.statut in ['closed', 'vendu']);
      allow delete: if isAdmin();

      match /offers/{offerId} {
        allow read: if true;
        allow create, update: if isAdmin()
                              || (request.auth != null
                                  && request.auth.uid == offerId
                                  && subscriptionValide(request.auth.uid));
        allow delete: if isAdmin();
      }
    }

    // Comptes magasins
    match /stores/{storeId} {
      allow create: if isOwner(storeId)
                      && request.resource.data.actif == false
                      && request.resource.data.subscriptionStatus == 'essai';
      // Admin lit tout ; le magasin ne lit que le sien
      allow read: if isAdmin() || isOwner(storeId);
      // Admin peut tout modifier (activation, abo, blocage)
      allow update: if isAdmin()
                    || (isOwner(storeId)
                        && request.resource.data.actif == resource.data.actif
                        && request.resource.data.subscriptionStatus
                            == resource.data.subscriptionStatus
                        && request.resource.data.trialEndDate
                            == resource.data.trialEndDate
                        && request.resource.data.get('subscriptionEndDate', null)
                            == resource.data.get('subscriptionEndDate', null)
                        && request.resource.data.get('currentPlanId', null)
                            == resource.data.get('currentPlanId', null));
      allow delete: if isAdmin();

      match /payment_requests/{paymentId} {
        allow create: if isOwner(storeId)
                        && request.resource.data.storeId == storeId
                        && request.resource.data.statut == 'en_attente';
        allow read: if isAdmin() || isOwner(storeId);
        // Admin valide / refuse les preuves de paiement
        allow update: if isAdmin();
        allow delete: if isAdmin();
      }
    }

    function subscriptionValide(uid) {
      let store = get(/databases/$(database)/documents/stores/$(uid)).data;
      return store.actif == true
        && (
          (store.subscriptionStatus == 'essai'
            && store.trialEndDate > request.time)
          || (store.subscriptionStatus == 'actif'
            && store.get('subscriptionEndDate', request.time) > request.time)
        );
    }
  }
}
